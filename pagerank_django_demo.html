from django.db import models class Node(models.Model): name = models.CharField(max_length=100, unique=True) pagerank_score = models.FloatField(default=0.0) created_at = models.DateTimeField(auto_now_add=True) def __str__(self): return f"{self.name} (PR={self.pagerank_score:.4f})" class Meta: ordering = ['-pagerank_score'] class Edge(models.Model): source = models.ForeignKey( Node, on_delete=models.CASCADE, related_name='outgoing_edges' ) target = models.ForeignKey( Node, on_delete=models.CASCADE, related_name='incoming_edges' ) weight = models.FloatField(default=1.0) class Meta: unique_together = ['source', 'target'] def __str__(self): return f"{self.source.name} → {self.target.name}"

""" PageRank implementation using the power iteration method. PR(u) = (1-d)/N + d * Σ PR(v) / OutDegree(v) """ from typing import Dict, List, Tuple from .models import Node, Edge def run_pagerank( damping: float = 0.85, max_iter: int = 100, tol: float = 1e-6 ) -> Tuple[Dict[str, float], int, List[float]]: """ Returns (scores_dict, iterations_used, convergence_errors) """ nodes = list(Node.objects.all()) edges = list(Edge.objects.select_related('source', 'target').all()) N = len(nodes) if N == 0: return {}, 0, [] # Build adjacency: out_links[node_id] = [target_id, ...] node_ids = {n.id: i for i, n in enumerate(nodes)} out_links = {i: [] for i in range(N)} for edge in edges: src = node_ids[edge.source_id] tgt = node_ids[edge.target_id] out_links[src].append(tgt) # Initialise uniform ranks ranks = [1.0 / N] * N errors = [] for iteration in range(max_iter): new_ranks = [(1.0 - damping) / N] * N for src_i, targets in out_links.items(): if not targets: # dangling node contrib = ranks[src_i] / N new_ranks = [r + damping * contrib for r in new_ranks] else: share = damping * ranks[src_i] / len(targets) for tgt_i in targets: new_ranks[tgt_i] += share error = sum(abs(new_ranks[i] - ranks[i]) for i in range(N)) errors.append(error) ranks = new_ranks if error < tol: # Save scores back to DB for i, node in enumerate(nodes): node.pagerank_score = ranks[i] node.save(update_fields=['pagerank_score']) return ( {nodes[i].name: ranks[i] for i in range(N)}, iteration + 1, errors ) raise RuntimeError("PageRank did not converge")

from django.http import JsonResponse from django.views import View from django.shortcuts import render from .models import Node, Edge from .pagerank import run_pagerank import json class GraphView(View): def get(self, request): return render(request, 'graph_app/index.html') class SeedGraphView(View): """POST: seed sample web-page graph into the DB""" def post(self, request): pages = ['Home', 'About', 'Blog', 'Contact', 'Docs', 'Forum', 'Search', 'Login'] links = [ ('Home','Blog'), ('Home','About'), ('Home','Docs'), ('Blog','Home'), ('Blog','Forum'), ('About','Home'), ('Docs','Home'), ('Docs','Forum'), ('Forum','Docs'), ('Forum','Blog'), ('Search','Home'),('Login','Home'), ('Contact','Home') ] Node.objects.all().delete() nodes = {name: Node.objects.create(name=name) for name in pages} for src, tgt in links: Edge.objects.create( source=nodes[src], target=nodes[tgt]) return JsonResponse({'status': 'seeded', 'nodes': len(pages), 'edges': len(links)}) class RunPageRankView(View): """POST: run algorithm, GET: return stored scores""" def post(self, request): body = json.loads(request.body or '{}') damping = float(body.get('damping', 0.85)) scores, iters, errors = run_pagerank(damping=damping) return JsonResponse({ 'scores': scores, 'iterations': iters, 'final_error': errors[-1], 'convergence': errors }) # graph_app/urls.py # urlpatterns = [ # path('', GraphView.as_view()), # path('api/seed/', SeedGraphView.as_view()), # path('api/pagerank/', RunPageRankView.as_view()), # ]
